    Schnittstelle DitherStrategy

    Strategy-Interface für Dithering-Algorithmen.

    Erhält ein Grauwert-Array (0=schwarz, 255=weiß) und liefert ein Schwellwert-Array (true=Dot drucken, false=leer).

    interface DitherStrategy {
        name: string;
        dither(gray: Uint8Array, width: number, height: number): Uint8Array;
    }
